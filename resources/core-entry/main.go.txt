package main

import (
	"context"
	"errors"
	"flag"
	"fmt"
	"os"
	"os/signal"
	"syscall"

	box "github.com/sagernet/sing-box"
	C "github.com/sagernet/sing-box/constant"
	"github.com/sagernet/sing-box/experimental/deprecated"
	"github.com/sagernet/sing-box/include"
	"github.com/sagernet/sing-box/log"
	"github.com/sagernet/sing-box/option"
	"github.com/sagernet/sing/common/json"
	"github.com/sagernet/sing/service"
)

func main() {
	if err := runMain(os.Args[1:]); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}

func runMain(args []string) error {
	if len(args) == 0 {
		return usageError()
	}
	switch args[0] {
	case "version":
		return runVersion(args[1:])
	case "check":
		configPath, err := parseConfigFlag("check", args[1:])
		if err != nil {
			return err
		}
		return checkConfig(configPath)
	case "run":
		configPath, err := parseConfigFlag("run", args[1:])
		if err != nil {
			return err
		}
		return runConfig(configPath)
	default:
		return fmt.Errorf("unknown command %q: %w", args[0], usageError())
	}
}

func usageError() error {
	return errors.New("usage: karing-tui-core <version|check|run> [-c config.json]")
}

func runVersion(args []string) error {
	flags := flag.NewFlagSet("version", flag.ContinueOnError)
	flags.SetOutput(os.Stderr)
	nameOnly := flags.Bool("n", false, "print version name only")
	if err := flags.Parse(args); err != nil {
		return err
	}
	if flags.NArg() != 0 {
		return usageError()
	}
	if *nameOnly {
		fmt.Println(C.Version)
	} else {
		fmt.Println("sing-box version " + C.Version)
	}
	return nil
}

func parseConfigFlag(name string, args []string) (string, error) {
	flags := flag.NewFlagSet(name, flag.ContinueOnError)
	flags.SetOutput(os.Stderr)
	configPath := flags.String("c", "", "configuration file path")
	if err := flags.Parse(args); err != nil {
		return "", err
	}
	if flags.NArg() != 0 || *configPath == "" {
		return "", errors.New(name + " requires exactly one -c configuration file")
	}
	return *configPath, nil
}

func coreContext() context.Context {
	ctx := context.Background()
	return include.Context(service.ContextWith(ctx, deprecated.NewStderrManager(log.StdLogger())))
}

func loadOptions(ctx context.Context, configPath string) (option.Options, error) {
	content, err := os.ReadFile(configPath)
	if err != nil {
		return option.Options{}, fmt.Errorf("read config: %w", err)
	}
	options, err := json.UnmarshalExtendedContext[option.Options](ctx, content)
	if err != nil {
		return option.Options{}, fmt.Errorf("decode config: %w", err)
	}
	return options, nil
}

func newInstance(configPath string) (*box.Box, context.CancelFunc, error) {
	ctx := coreContext()
	options, err := loadOptions(ctx, configPath)
	if err != nil {
		return nil, nil, err
	}
	instanceCtx, cancel := context.WithCancel(ctx)
	instance, err := box.New(box.Options{
		Context: instanceCtx,
		Options: options,
	})
	if err != nil {
		cancel()
		return nil, nil, fmt.Errorf("create core: %w", err)
	}
	return instance, cancel, nil
}

func checkConfig(configPath string) error {
	instance, cancel, err := newInstance(configPath)
	if err != nil {
		return err
	}
	defer cancel()
	if err := instance.Close(); err != nil {
		return fmt.Errorf("close checked core: %w", err)
	}
	return nil
}

func runConfig(configPath string) error {
	instance, cancel, err := newInstance(configPath)
	if err != nil {
		return err
	}
	defer cancel()

	if err := instance.Start(); err != nil {
		_ = instance.Close()
		return fmt.Errorf("start core: %w", err)
	}

	signals := make(chan os.Signal, 1)
	signal.Notify(signals, os.Interrupt, syscall.SIGTERM)
	defer signal.Stop(signals)

	<-signals
	cancel()
	if err := instance.Close(); err != nil {
		return fmt.Errorf("close core: %w", err)
	}
	return nil
}
